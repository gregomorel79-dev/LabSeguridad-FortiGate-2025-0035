# El resto de la configuracion se hizo por GUI.

# 1) Acceso de gestion inicial por port1
config system interface
edit port1
set allowaccess ping https http ssh
set defaultgw disable
next
# 2) WAN por DHCP para activar la licencia de evaluacion
edit port2
set mode dhcp
set allowaccess ping
next
end

config router static
edit 1
set device port2
set gateway 192.168.42.1
next
end

# 3) Liberar una interfaz (limite de 3 en la licencia trial)
config system dhcp server
delete 1
end
config system ntp
set server-mode disable
end
config system interface
delete fortilink
end

# 4) Mover la gestion a IP secundaria de port3 y dedicar port1 a la VLAN10
config system interface
edit port1
set mode static
set ip 10.0.35.129 255.255.255.128
set allowaccess ping
set alias USUARIOS-VLAN10
next
edit port3
set ip 10.0.35.1 255.255.255.240
set allowaccess ping https ssh http
set secondary-IP enable
config secondaryip
edit 1
set ip 192.168.56.50 255.255.255.0
set allowaccess ping https ssh http
next
end
next
end

# 5) Regla IPS adicional por filtro (SQLi generico)
config ips sensor
edit Anti-SQLi
config entries
edit 2
set protocol HTTP
set location server
set severity medium high critical
set status enable
set log-packet enable
set action block
set quarantine attacker
next
end
next
end
